CLI Playbook: Chuyển đổi 2 Luồng 
# Luồng 1 (Mặc định): Full Cloud Qdrant (1,408 MB RAM)
docker compose --profile core up -d --build

# Luồng 2 (Dự phòng): Full Local Backup (1,920 MB RAM) + Nạp lại từ Cloudflare R2
$env:QDRANT_MODE = "local"
docker compose --profile core --profile local-backup up -d
uv run python -m acdp.v2.rag_ingest
